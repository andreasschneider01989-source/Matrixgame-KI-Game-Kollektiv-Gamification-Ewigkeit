	// --- Konstruktor --- init1
	public MainGame() {
		// Canvas mit schwarzem Hintergrund erstellen
		setPreferredSize(new Dimension(FRAME_WIDTH, FRAME_HEIGHT));
		setBackground(Color.BLACK);

		MouseAdapter mouseHandler = new MouseAdapter() {
			@Override
			public void mousePressed(MouseEvent e) {
				// 1. Linksklick: Einheiten selektieren
				if (SwingUtilities.isLeftMouseButton(e)) {
					int mouseX = e.getX();
					int mouseY = e.getY();
					boolean unitFound = false;

					for (Unit u : unitList) {
						double sx = (zoomCenterX + u.posX) * zoomFactor + getWidth() / 2.0;
						double sy = (zoomCenterY + u.posY) * zoomFactor + getHeight() / 2.0;
						double radius = Math.abs(u.size);

						if (Math.hypot(mouseX - sx, mouseY - sy) <= radius) {
							u.isSelected = true;
							System.out.println("Unit " + u.id + " Wurde vom Spieler ausgewählt");
							unitFound = true;
						} else if (!e.isShiftDown()) {
							u.isSelected = false;
						}
					}

					if (!unitFound && !e.isShiftDown()) {
						for (Unit u : unitList) {
							u.isSelected = false;
						}
					}
					repaint();
				}
				// 2. Rechtsklick: Befehle an ausgewählte Einheiten verteilen
				// 2. Rechtsklick: Befehle an ausgewählte Einheiten verteilen
				else if (SwingUtilities.isRightMouseButton(e)) {// 2. Rechtsklick: Befehle an ausgewählte Einheiten
																// verteilen

					int mouseX = e.getX();
					int mouseY = e.getY();

					// KORRIGIERTE MATHEMATISCHE INVERSION: Screen -> World
					double worldTargetX = (mouseX - getWidth() / 2.0) / zoomFactor - zoomCenterX;
					double worldTargetY = (mouseY - getHeight() / 2.0) / zoomFactor - zoomCenterY;

					System.out.println("--- Befehlsausgabe (Rechtsklick) ---");
					System.out.println("Maus Screen-X: " + mouseX + " | Screen-Y: " + mouseY);
					System.out.println("Berechnete Welt-X: " + worldTargetX + " | Welt-Y: " + worldTargetY);

					Unit clickedEnemy = null;
					for (Unit u : unitList) {
						// Hier bleibt die World -> Screen Transformation korrekt,
						// da wir prüfen, wo die Einheit auf dem Bildschirm gerendert wird.
						double sx = (zoomCenterX + u.posX) * zoomFactor + getWidth() / 2.0;
						double sy = (zoomCenterY + u.posY) * zoomFactor + getHeight() / 2.0;
						double radius = Math.abs(u.size);

						if (Math.hypot(mouseX - sx, mouseY - sy) <= radius) {
							clickedEnemy = u;
							System.out.println("Gegnerische Einheit für Angriff anvisiert: ID " + u.id);
							break;
						}
					}

					for (Unit u : unitList) {
						if (u.isSelected) {
							if (clickedEnemy != null) {
								// Angriffsbefehl
								u.manualTarget = clickedEnemy;
								u.targetPosX = null;
								u.targetPosY = null;
								u.decisionAlgorithm = "TARGET_ENEMY";
							} else {
								// Bewegungs-Befehl
								u.targetPosX = worldTargetX;
								u.targetPosY = worldTargetY;
								u.manualTarget = null;
								u.decisionAlgorithm = "TARGET_POINT";
								System.out.println("Einheit ID " + u.id + " bewegt sich zu Zielpunkt.");
							}
						}
					}
					repaint();
				}
			}
		};

		this.addMouseListener(mouseHandler);
		this.addMouseMotionListener(mouseHandler);
	}

	// Gelogene Methode
	@Override
	protected void paintComponent(Graphics g) {
		super.paintComponent(g);
		Graphics2D g2 = (Graphics2D) g;
		g2.setColor(Color.BLACK);
		g2.fillRect(0, 0, getWidth(), getHeight());

		// Loop für alle Units aus UnitList
		for (int i = 0; i < unitList.size(); i++) {
			Unit unit = unitList.get(i);
			if (unit.leben <= 0)
				continue;
			double radius = unit.size;
			Color color = new Color((int) (unit.red), (int) (unit.green), (int) (unit.blue));
			double posX = unit.posX;
			double posY = unit.posY;
			// Berücksichtige Pan und Zoom bei der Positionierung
			double sx = (zoomCenterX + posX) * zoomFactor + getWidth() / 2.0;
			double sy = (zoomCenterY + posY) * zoomFactor + getHeight() / 2.0;
//			int r = (int) Math.max(body.radius * zoomFactor, MIN_RADIUS_DRAW);
			g2.setColor(color);
			g2.fillOval((int) (sx - radius), (int) (sy - radius), (int) (radius * 2), (int) (radius * 2));
			// Innerhalb von paintComponent(Graphics g) in der for-Schleife:
			if (unit.isSelected) {
				g2.setColor(Color.WHITE);
				// Einen etwas größeren Ring um die Unit zeichnen
				g2.drawOval((int) (sx - radius - 2), (int) (sy - radius - 2), (int) (radius * 2 + 4),
						(int) (radius * 2 + 4));
			}

		}
	}

	// Zoom-Funktionen
	public void zoomIn() {
		zoomFactor *= ZOOM_STEP;
		PAN_STEP = feinheit / zoomFactor;

	}

	public void zoomOut() {
		zoomFactor /= ZOOM_STEP;

	}

	long pasueTime = 10;
	long kompletteRechenzeitDerSimulation = 0;
	double avgRechenzeitProUnitProEinheitenZug = 0;
	double count = 0;

	public void gameLoop() {
		while (true) {
			repaint();

			if (unitList.isEmpty())
				break; // Sicherheitscheck

			int randNumber = (int) (unitList.size() * Math.random());
			Unit current = unitList.get(randNumber);

			if (current.leben <= 0) {
				if (current.typ != null && current.typ.equals("King")) {
					System.out.println("--- GAME OVER ---");
					System.out.println("Der König von Team " + current.team + " wurde vernichtet!");
					System.out.println("Das andere Team hat gewonnen!");
					return;
				} else {
					continue; // Tote Unit überspringen
				}
			}

			current.leben *= 1.1;
			if (current.leben > current.maxLeben)
				current.leben = current.maxLeben;
			current.currentDefense = current.defenseProTurn;

			// --- 🚀 OPTIMIERUNG: DYNAMISCHES LOAD-BALANCING (SKIP LOGIC) ---
			// Wir warten bis count > 50, damit sich der Durchschnitt erst mal einpendelt.
			// Wenn eine Unit 50% mehr Zeit braucht als der Durchschnitt, wird sie mit 50%
			// Wahrscheinlichkeit diesen Tick "übersprungen".
			// Das spart CPU und simuliert z.B. "langsamere Reaktionszeiten" bei komplexen
			// Einheiten.

			double zugChance = avgRechenzeitProUnitProEinheitenZug / current.rechenzeitProZug;

			if (Math.random() > zugChance) {
				continue; // Skip! Die Unit behält ihren aktuellen Bewegungswert (vx, vy) bei.
			}

			// current.printWerte(); // TIPP: Für maximale Performance später
			// auskommentieren!

			long startTime = System.nanoTime();
			count++;

			// --- ENTSCHEIDUNGSLOGIK ---
			if ("TARGET_POINT".equals(current.decisionAlgorithm) && current.targetPosX != null
					&& current.targetPosY != null) {

				current.moveTo(current.targetPosX, current.targetPosY, current.posZ, current.maxSpeed);
				double distToTarget = current.getDistanceTo(current.targetPosX, current.targetPosY, current.posZ);
				if (distToTarget < 1.5) {
					current.decisionAlgorithm = "NEAREST_ENEMY";
					current.targetPosX = null;
					current.targetPosY = null;
				}
			} else if ("TARGET_ENEMY".equals(current.decisionAlgorithm) && current.manualTarget != null) {
				if (current.manualTarget.leben <= 0) {
					current.decisionAlgorithm = "NEAREST_ENEMY";
					current.manualTarget = null;
				} else {
					current.target = current.manualTarget;
					if (current.getDistance(current.target) <= current.reichweite) {
						current.attackieren(current.target);
					} else {
						current.moveTo(current.target);
					}
				}
			} else {

				//////
				// Standard KI
				if (current.target == null || current.target.leben <= 0) {
					current.target = nextGegner(current);
				}
				if (current.getDistance(current.target) > current.reichweite + current.maxSpeed)
					current.target = nextGegner(current);
				current.update();

				if (current.target.typ != null && current.target.typ.equals("King") && current.target.leben <= 0) {
					System.out.println("Der König von Team " + current.target.team + " wurde vernichtet!");
					System.out.println("Team " + current.team + " hat gewonnen!");
					return;
				}

			}

			long endTime = System.nanoTime();
			long zugZeit = endTime - startTime;

			// Zeitmessung speichern (Gleitender Durchschnitt für stabile Werte)
			current.rechenzeitProZug = (current.rechenzeitProZug * 0.8) + (zugZeit * 0.2);
			current.kompletteRechenzeitInSimulation += zugZeit;
			kompletteRechenzeitDerSimulation += zugZeit;
			avgRechenzeitProUnitProEinheitenZug = kompletteRechenzeitDerSimulation / count;

			// NEU: Kurze Pause (16ms = ca. 60 FPS Update-Rate für die Logik)
			try {
				Thread.sleep(pasueTime);
			} catch (InterruptedException e) {
				e.printStackTrace();
			}
		}
	}

	public Unit nextGegner(Unit current) {
		Unit gegner = null;
		double min = 999999999;
		for (Unit ziel : unitList) {
			if (current.getDistance(ziel) < min && current.team != ziel.team && ziel.leben > 0) {
				min = current.getDistance(ziel);
				gegner = ziel;
			}
		}
		return gegner;
	}

	public void moveTo(KreisObjekt oponent) {
		double distQ = (this.posX - oponent.posX) * (this.posX - oponent.posX)
				+ (this.posY - oponent.posY) * (this.posY - oponent.posY)
				+ (this.posZ - oponent.posZ) * (this.posZ - oponent.posZ);
		double dist = Math.pow(distQ, 0.5);

		double dx = posX - oponent.posX;
		double dy = posY - oponent.posY;
		double dz = posZ - oponent.posZ;

		double movDist = Math.min(maxSpeed * ((System.currentTimeMillis() - timeBefore) / 1000.0),
				Math.abs(dist - reichweite));
		movDist /= 1.01;

		vx = -dx / dist * movDist;
		vy = -dy / dist * movDist;
		vz = -dz / dist * movDist;

		System.out.println("Move: " + vx + "," + vy + "," + vz);

		move(1);

	}

	public void moveToAndKite(KreisObjekt oponent) {
		double distQ = (this.posX - oponent.posX) * (this.posX - oponent.posX)
				+ (this.posY - oponent.posY) * (this.posY - oponent.posY)
				+ (this.posZ - oponent.posZ) * (this.posZ - oponent.posZ);
		double dist = Math.pow(distQ, 0.5);

		double dx = posX - oponent.posX;
		double dy = posY - oponent.posY;
		double dz = posZ - oponent.posZ;

		double movDist = Math.min(maxSpeed,
				Math.abs(dist - reichweite));

		if (reichweite > dist) {
			movDist = reichweite - dist;
			movDist = -movDist;}
		
		
			if(movDist>0)movDist = movDist+0.01;
			else movDist= movDist+0.01;
			
		
		

		vx = -dx / dist * movDist;
		vy = -dy / dist * movDist;
		vz = -dz / dist * movDist;

		System.out.println("Move: " + vx + "," + vy + "," + vz);

		move(1);

	}

		}

		if (!legalMove()) {
			posX = posXBefore;
			posY = posYBefore;
			posZ = posZBefore;
		}

		// Angriff nur wenn bereit und in Reichweite
		if (this.target != null) if( getDistance(this.target) <= reichweite) {
			if( canAttack()) {
				performAttack(this.target); // Neue Methode verwenden
			}} else System.out.println("Reichweite  ist zu klein: Reichweite = "+reichweite+"Abstand = "+getDistance(this.target));
		

		posXBefore = posX;
		posYBefore = posY;
		posZBefore = posZ;
		timeBefore = currentTime;

	}

	public void moveTo(double posX, double posY, double posZ, double speedManuel) {
		System.out.println("Zug = moveTo(" + posX + "," + posY + "," + posZ + "," + speedManuel + ")");
		double distQ = (this.posX - posX) * (this.posX - posX) + (this.posY - posY) * (this.posY - posY)
				+ (this.posZ - posZ) * (this.posZ - posZ);

		// FEHLER 3 BEHOBEN: targetPosX hier NICHT auf null setzen!
		// Das macht der MainGame-Loop sicherer.

		double dist = Math.pow(distQ, 0.5);

		// Verhindert Division durch Null, falls wir exakt auf dem Punkt stehen
		if (dist < 0.001) {
			vx = 0;
			vy = 0;
			vz = 0;
			move(1);
			return;
		}

		double dx = this.posX - posX;
		double dy = this.posY - posY;
		double dz = this.posZ - posZ;

		double movDist = Math.min(speedManuel, Math.min(maxSpeed, Math.abs(dist - 0.0000005)));

		vx = -dx / dist * movDist;
		vy = -dy / dist * movDist;
		vz = -dz / dist * movDist;

		System.out.println("Vektor = " + vx + "," + vy + "," + vz);

		// FEHLER 1 BEHOBEN: Die Bewegung muss ausgeführt werden!
		move(1);
	}

	// entropieMove das Objekt kann sich hier per Zufall in jeder Positionsvariable
	// incrementiert oder decrementiert werden. Die Summe aller Differenzen
	// multipliziert mit ihrer Wahrscheinlichkeit kann nicht größer als die
	// Geschwindigkeit sein bzw. es darf nicht möglich sein, das sich ein Objekt per
	// Zufall pro Zeitschritt in einer Art und Weise verändern kann die für die
	// moveTo Methode zu schnell bzw. zu weit wäre
	// Zufälliges Incrementieren oder decrementieren der x Position
	public void entropieMoveX1(double px, double pxInv) {
		if (px > Math.random())
			this.posX++;
		if (pxInv > Math.random())
			this.posX--;
	}

	// Zufälliges Incrementieren oder decrementieren der y Position
	public void entropieMoveY1(double py, double pyInv) {
		if (py > Math.random())
			this.posY++;
		if (pyInv > Math.random())
			this.posY--;
	}

	// Zufälliges Incrementieren oder decrementieren der z Position
	public void entropieMoveZ1(double pz, double pzInv) {
		if (pz > Math.random())
			this.posZ++;
		if (pzInv > Math.random())
			this.posZ--;
	}

	public void entropieMove(double px, double pxInv, double py, double pyInv, double pz, double pzInv) {
		entropieMoveX1(px, pxInv);
		entropieMoveX1(py, pyInv);
		entropieMoveX1(pz, pzInv);
	}

	public void moveTrigonometric(Unit target) {
		if (target == null || target.leben <= 0)
			return;

		double dx = target.posX - this.posX;
		double dy = target.posY - this.posY;
		double dist = Math.hypot(dx, dy);

		if (dist < 0.001)
			return;

		double angle = Math.atan2(dy, dx);
		double moveAngle = angle;

		// TAKTIK: Wenn wir Fernkämpfer sind, aber der Feind zu nah kommt ->
		// Kiting/Orbit
		if (this.reichweite > 3.0 && dist < this.reichweite * 1.5) {
			// Wir laufen nicht direkt weg, sondern seitlich (Flanken)
			moveAngle = angle + (Math.PI / 2.5);
		}

		this.vx = Math.cos(moveAngle) * this.maxSpeed;
		this.vy = Math.sin(moveAngle) * this.maxSpeed;

		move(1);
	}

	public double getDistance(Unit oponent) {
		double distQ = (this.posX - oponent.posX) * (this.posX - oponent.posX)
				+ (this.posY - oponent.posY) * (this.posY - oponent.posY)
				+ (this.posZ - oponent.posZ) * (this.posZ - oponent.posZ);
		double dist = Math.pow(distQ, 0.5);
		return dist;
	}

	public double getDistanceTo(double x, double y, double z) {
		double distQ = (this.posX - x) * (this.posX - x) + (this.posY - y) * (this.posY - y)
				+ (this.posZ - z) * (this.posZ - z);
		double dist = Math.pow(distQ, 0.5);
		return dist;
	}

	/**
	 * Prüft ob die Unit angreifen darf basierend auf attackSpeed
	 * 
	 * @return true wenn Angriff bereit, false wenn noch Cooldown
	 */
	public boolean canAttack() {
		long currentTime = System.currentTimeMillis();

		// Wenn noch nie angegriffen, sofort bereit
		if (lastAttackTime == 0) {
			return true;
		}

		// Berechne benötigte Wartezeit in Millisekunden
		double cooldownMillis = (1.0 / attackSpeed) * 1000.0;

		// Prüfe ob Cooldown abgelaufen ist
		return (currentTime - lastAttackTime) >= cooldownMillis;
	}
	
	public boolean canMove() {
		long currentTime = System.currentTimeMillis();

		// Wenn noch nie bewegt, sofort bereit
		if (lastMoveTime == 0) {
			return true;
		}

		// Berechne benötigte Wartezeit in Millisekunden
		double cooldownMillis = (1.0 / moveSpeed) * 1000.0;

		// Prüfe ob Cooldown abgelaufen ist
		return (currentTime - lastMoveTime) >= cooldownMillis;
	}

	/**
	 * Führt Angriff aus und setzt Timestamp
	 */
	public void performAttack(Unit target) {
		if (canAttack()) {
			// Angriff ausführen
			attackieren(target);

			// Timestamp aktualisieren
			lastAttackTime = System.currentTimeMillis();
		} else {
			System.out.println("Illegaler Zug");
			// Optional: Debug-Ausgabe
			double remainingCooldown = ((1.0 / attackSpeed) * 1000.0) - (System.currentTimeMillis() - lastAttackTime);
			System.out.println("Unit " + id + " muss noch " + String.format("%.2f", remainingCooldown / 1000.0)
					+ "s warten bis nächster Angriff");
		}
	}

	public void performMove() {

	public void attackieren(Unit target) {

		if (getDistance(target) > reichweite)
			return;
//			{moveTo(target);System.out.println("Zug = moveTo("+target.id+")");}
		else {

			if (legalMove()) {
				System.out.println("Zug = attackieren(" + target.id + ")");
				double damage = this.damage;
				if (this.kritchance > Math.random()) {
					double rand = Math.random();
					System.out.println("Kritischer Treffer! Schaden Multipliziert mit " + 1.0 / rand);
					damage /= rand;
				}
				if (target.currentDefense * Math.random() > this.geschicklichkeit * Math.random()) {
					target.currentDefense -= (this.geschicklichkeit / Math.random());
					System.out.println("Attacke wurde geblockt und hat keinen Schaden gemacht");
					return;
				}
				target.leben = target.leben - this.damage;
				if (target.leben <= 0) {

					System.out.println("Unit " + target.id + " Wurde vernichtet durch Unit " + this.id);
					
				}
			}
		}
	}

	public boolean legalMove() {
		currentTime = System.currentTimeMillis();
		deltaTime = currentTime - timeBefore;
		timeSeconds = deltaTime / 1000.0;
		timeBefore = currentTime;

		double deltaX = posX - posXBefore;
		double deltaY = posY - posYBefore;
		double deltaZ = posZ - posZBefore;

		double distance = Math.sqrt(deltaX * deltaX + deltaY * deltaY + deltaZ * deltaZ);

		// FIX: Nur prüfen wenn genug Zeit vergangen ist (mindestens 1ms)
		if (timeSeconds > 0.001) {
			double geschwindigkeit = distance / timeSeconds;
			// FIX: 20% Toleranz hinzufügen
			if (geschwindigkeit > maxSpeed*moveSpeed) {
				System.out.println("Illegaler Zug (zu schnell): " + geschwindigkeit + " > " + maxSpeed);
				return false;
			}
		}

		// Grid-Prüfung bleibt
		if (grid) {
			if (deltaX != (int) (Math.round(deltaX)))
				return false;
			if (deltaY != (int) (Math.round(deltaY)))
				return false;
			if (deltaZ != (int) (Math.round(deltaZ)))
				return false;
		}

		return true;
	}

	boolean grid;

	// Geschwindigkeitsvektoren

	public boolean isSelected = false;

	public KreisObjekt(double size, double red, double green, double blue, double posX, double posY, double posZ,
			double vx, double vy, double vz) {

	// Default mit interessantem Random (für die Initialisierung des Kindes)
	public KreisObjekt() {
		this.posX = 0;
		this.posY = 0;
		this.posZ = 0;
		this.vx = 0;
		this.vy = 0;
		this.vz = 0;
		this.size = 10;
	}

	// Mathematische Vektor-Addition für den Zeitschritt (Kausalkette)
	public void move(double dt) {
		

		
		if (grid) {
			// Grid-Logik (Snapping)
			if (Math.abs(this.vx) < 0.5 && Math.abs(this.vy) < 0.5) {
				if (Math.abs(this.vx) > Math.abs(this.vy)) {
					vx = (this.vx > 0) ? 1 : -1; vy = 0;
				} else {
					vy = (this.vy > 0) ? 1 : -1; vx = 0;
				}
			}
			this.posX += this.vx * dt;
			this.posY += this.vy * dt;
			this.posZ += this.vz * dt;

			this.posX = Math.round(this.posX);
			this.posY = Math.round(this.posY);
			this.posZ = Math.round(this.posZ);
		} else {
			// FREIE BEWEGUNG (Der fehlende Else-Zweig!)
			this.posX += this.vx * dt;
			this.posY += this.vy * dt;
			this.posZ += this.vz * dt;
		}
		
		
		
	}
}

	// --- Pan-Methoden ---
	public void panLeft() {
		zoomCenterX += PAN_STEP;
		repaint();
	}

	public void panRight() {
		zoomCenterX -= PAN_STEP;
		repaint();
	}

	public void panUp() {
		zoomCenterY += PAN_STEP;
		repaint();
	}

	public void panDown() {
		zoomCenterY -= PAN_STEP;
		repaint();
	}

	// Gelogene Methode
	@Override
	protected void paintComponent(Graphics g) {
		super.paintComponent(g);
		Graphics2D g2 = (Graphics2D) g;
		g2.setColor(Color.BLACK);
		g2.fillRect(0, 0, getWidth(), getHeight());

		// Loop für alle Units aus UnitList
		for (int i = 0; i < unitList.size(); i++) {
			Unit unit = unitList.get(i);
			if (unit.leben <= 0)
				continue;
			double radius = unit.size;
			Color color = new Color((int) (unit.red), (int) (unit.green), (int) (unit.blue));
			double posX = unit.posX;
			double posY = unit.posY;
			// Berücksichtige Pan und Zoom bei der Positionierung
			double sx = (zoomCenterX + posX) * zoomFactor + getWidth() / 2.0;
			double sy = (zoomCenterY + posY) * zoomFactor + getHeight() / 2.0;
//			int r = (int) Math.max(body.radius * zoomFactor, MIN_RADIUS_DRAW);
			g2.setColor(color);
			g2.fillOval((int) (sx - radius), (int) (sy - radius), (int) (radius * 2), (int) (radius * 2));
			// Innerhalb von paintComponent(Graphics g) in der for-Schleife:
			if (unit.isSelected) {
				g2.setColor(Color.WHITE);
				// Einen etwas größeren Ring um die Unit zeichnen
				g2.drawOval((int) (sx - radius - 2), (int) (sy - radius - 2), (int) (radius * 2 + 4),
						(int) (radius * 2 + 4));
			}

		}
	}

	// Zoom-Funktionen
	public void zoomIn() {
		zoomFactor *= ZOOM_STEP;
		PAN_STEP = feinheit / zoomFactor;
		repaint();
	}

	public void zoomOut() {
		zoomFactor /= ZOOM_STEP;
		PAN_STEP = feinheit / zoomFactor;
		repaint();
	}

	long pasueTime = 0;
	long kompletteRechenzeitDerSimulation = 0;
	double avgRechenzeitProUnitProEinheitenZug = 0;
	double count = 0;

	public void gameLoop() {
		while (true) {
			repaint();

			if (unitList.isEmpty())
				break; // Sicherheitscheck

			int randNumber = (int) (unitList.size() * Math.random());
			Unit current = unitList.get(randNumber);

			if (current.leben <= 0) {
				if (current.typ != null && current.typ.equals("King")) {
					System.out.println("--- GAME OVER ---");
					System.out.println("Der König von Team " + current.team + " wurde vernichtet!");
					System.out.println("Das andere Team hat gewonnen!");
					return;
				} else {
					continue; // Tote Unit überspringen
				}
			}

			current.leben *= 1.1;
			if (current.leben > current.maxLeben)
				current.leben = current.maxLeben;
			current.currentDefense = current.defenseProTurn;

			// --- 🚀 OPTIMIERUNG: DYNAMISCHES LOAD-BALANCING (SKIP LOGIC) ---
			// Wir warten bis count > 50, damit sich der Durchschnitt erst mal einpendelt.
			// Wenn eine Unit 50% mehr Zeit braucht als der Durchschnitt, wird sie mit 50%
			// Wahrscheinlichkeit diesen Tick "übersprungen".
			// Das spart CPU und simuliert z.B. "langsamere Reaktionszeiten" bei komplexen
			// Einheiten.

			double zugChance = avgRechenzeitProUnitProEinheitenZug / current.rechenzeitProZug;

			if (Math.random() > zugChance) {
				continue; // Skip! Die Unit behält ihren aktuellen Bewegungswert (vx, vy) bei.
			}

			// current.printWerte(); // TIPP: Für maximale Performance später
			// auskommentieren!

			long startTime = System.nanoTime();
			count++;

			// --- ENTSCHEIDUNGSLOGIK ---
			if ("TARGET_POINT".equals(current.decisionAlgorithm) && current.targetPosX != null
					&& current.targetPosY != null) {

				current.moveTo(current.targetPosX, current.targetPosY, current.posZ, current.maxSpeed);
				double distToTarget = current.getDistanceTo(current.targetPosX, current.targetPosY, current.posZ);
				if (distToTarget < 1.5) {
					current.decisionAlgorithm = "NEAREST_ENEMY";
					current.targetPosX = null;
					current.targetPosY = null;
				}
			} else if ("TARGET_ENEMY".equals(current.decisionAlgorithm) && current.manualTarget != null) {
				if (current.manualTarget.leben <= 0) {
					current.decisionAlgorithm = "NEAREST_ENEMY";
					current.manualTarget = null;
				} else {
					current.target = current.manualTarget;
					if (current.getDistance(current.target) <= current.reichweite) {
						current.attackieren(current.target);
					} else {
						current.moveTo(current.target);
					}
				}
			} else {

				//////
				// Standard KI
				current.pPrint=0.01+current.kills*0.01;
				if (current.target == null || current.target.leben <= 0) {
					current.target = nextGegner(current);
				}
				if (current.getDistance(current.target) > current.reichweite + current.maxSpeed)
					current.target = nextGegner(current);
				current.update();

				if (current.target.typ != null && current.target.typ.equals("King") && current.target.leben <= 0) {
					System.out.println("Der König von Team " + current.target.team + " wurde vernichtet!");
					System.out.println("Team " + current.team + " hat gewonnen!");
					return;
				}

			}

			long endTime = System.nanoTime();
			long zugZeit = endTime - startTime;

			// Zeitmessung speichern (Gleitender Durchschnitt für stabile Werte)
			current.rechenzeitProZug = (current.rechenzeitProZug * 0.8) + (zugZeit * 0.2);
			current.kompletteRechenzeitInSimulation += zugZeit;
			kompletteRechenzeitDerSimulation += zugZeit;
			avgRechenzeitProUnitProEinheitenZug = kompletteRechenzeitDerSimulation / count;

			// NEU: Kurze Pause (16ms = ca. 60 FPS Update-Rate für die Logik)
			try {
				Thread.sleep(pasueTime);
			} catch (InterruptedException e) {
				e.printStackTrace();
			}
		}
	}

	public Unit nextGegner(Unit current) {
		Unit gegner = null;
		double min = 999999999;
		for (Unit ziel : unitList) {
			if (current.getDistance(ziel) < min && current.team != ziel.team && ziel.leben > 0) {
				min = current.getDistance(ziel);
				gegner = ziel;
			}
		}
		return gegner;
	}
