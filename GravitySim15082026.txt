	public void EntrophieSpeed(double alpha) {
	    int n = bodies.size();
	    if (n < 2) return;

	    for (int i = 0; i < n; i++) {
	        MasseKugel body1 = bodies.get(i);

	        // Zufällige Auswahl eines zweiten Körpers für die Interaktion
	        int j = (int) (Math.random() * n);
	        while (i == j) { // Sicherstellen, dass nicht derselbe Körper gewählt wird
	            j = (int) (Math.random() * n);
	        }
	        MasseKugel body2 = bodies.get(j);

	        // Berechnung der mittleren Geschwindigkeit für die Angleichung
	        double avg_vx = (body1.vx + body2.vx) / 2;
	        double avg_vy = (body1.vy + body2.vy) / 2;
	        double avg_vz = (body1.vz + body2.vz) / 2;

	        // Entropie-basierte Geschwindigkeitsanpassung
	        // Die Geschwindigkeiten werden proportional zur Differenz zur mittleren Geschwindigkeit angepasst.
	        // Die Konstante alpha steuert die Stärke der Angleichung.
	        body1.vx += (avg_vx - body1.vx) * alpha;
	        body1.vy += (avg_vy - body1.vy) * alpha;
	        body1.vz += (avg_vz - body1.vz) * alpha;

	        body2.vx += (avg_vx - body2.vx) * alpha;
	        body2.vy += (avg_vy - body2.vy) * alpha;
	        body2.vz += (avg_vz - body2.vz) * alpha;
	    }
	}

	public void doppelsternBasis() {
		// --- Doppelsternsystem (Zirkumbinäres System) ---
		// Eine Simulation von zwei Sternen, die einander umkreisen (Baryzentrum), 
		// mit inneren Planeten und einem äußeren "Matrix-Wächter"-Planeten.

		double massSternA = 380000;  // Etwas massereicher als unsere Sonne
		double massSternB = 220000;  // Etwas masseärmer (z.B. Roter Zwerg)
		double gesamtMasseSterne = massSternA + massSternB;
		
		double abstandSterne = 80.0; // Distanz der Sterne zueinander

		// 1. Berechnung des gemeinsamen Schwerpunktes (Baryzentrum)
		double rA = abstandSterne * (massSternB / gesamtMasseSterne);
		double rB = abstandSterne * (massSternA / gesamtMasseSterne);

		// 2. Berechnung der Orbitalgeschwindigkeiten der Sterne
		// Basierend auf Newtons Gravitationsgesetz
		double vOrbitalSterne = Math.sqrt(G * gesamtMasseSterne / abstandSterne);
		double vA = vOrbitalSterne * (rA / abstandSterne);
		double vB = vOrbitalSterne * (rB / abstandSterne);

		// --- Himmelskörper definieren ---
		
		// Stern A (Zentrum links)
		MasseKugel sternA = new MasseKugel(-rA, 0, 0, 0, -vA, 0, massSternA, 12, new Color(255, 220, 50));
		
		// Stern B (Zentrum rechts)
		MasseKugel sternB = new MasseKugel(rB, 0, 0, 0, vB, 0, massSternB, 9, new Color(255, 80, 50));

		// Planet 1: Erdähnlicher Planet, der NUR Stern A umkreist
		double distPlanet1 = 12.0; 
		double vPlanet1 = Math.sqrt(G * massSternA / distPlanet1);
		// Geschwindigkeit addiert sich zur Eigenbewegung von Stern A
		MasseKugel planet1 = new MasseKugel(-rA - distPlanet1, 0, 0, 0, -vA - vPlanet1, 0, 1.0, 5, Color.GREEN);

		// Planet 2: Heißer Gasriese, der NUR Stern B umkreist
		double distPlanet2 = 18.0;
		double vPlanet2 = Math.sqrt(G * massSternB / distPlanet2);
		MasseKugel planet2 = new MasseKugel(rB + distPlanet2, 0, 0, 0, vB + vPlanet2, 0, 250.0, 7, new Color(150, 100, 255));

		// Planet 3: Zirkumbinärer Planet (Umkreist BEIDE Sterne weit außen)
		// Masseschwerpunkt für ihn ist näherungsweise die Gesamtmasse beider Sterne
		double distPlanet3 = 250.0;
		double vPlanet3 = Math.sqrt(G * gesamtMasseSterne / distPlanet3);
		MasseKugel planet3 = new MasseKugel(0, distPlanet3, 0, -vPlanet3, 0, 0, 8.5, 6, new Color(100, 255, 200));

		// Alle Körper zur Simulation hinzufügen
		bodies.add(sternA);
		bodies.add(sternB);
		bodies.add(planet1);
		bodies.add(planet2);
		bodies.add(planet3);
	}
	
	
	public void galaktischesZentrumBasis() {

		// und extremen stellaren Orbits. 

		// Skalierte Masse für Sagittarius A* (Real ca. 4 Millionen Sonnenmassen)
		// Die Masse wird hier so skaliert, dass sie in die bestehende G-Konstante 
		// und die Timesteps der Simulation passt, ohne Floating-Point-Overflows zu erzeugen.
		double massSgrA = 1.5e9; 
		
		// Das Schwarze Loch ruht (näherungsweise) im Zentrum
		MasseKugel sgrA = new MasseKugel(0, 0, 0, 0, 0, 0, massSgrA, 15, Color.BLACK);

		// --- S-Sterne (Extrem elliptische Orbits) ---
		// Um elliptische Bahnen zu erzeugen, geben wir den Sternen an ihrem fernsten Punkt (Apoapsis)
		// eine Geschwindigkeit, die deutlich UNTER der perfekten Kreisbahngeschwindigkeit liegt.

		// Stern S2 (ein realer Stern, der Sgr A* umkreist)
		double distS2 = 350.0;
		double vKreisS2 = Math.sqrt(G * massSgrA / distS2);
		double vElliptischS2 = vKreisS2 * 0.45; // Nur 45% Speed -> starke Exzentrizität
		MasseKugel sternS2 = new MasseKugel(-distS2, 0, 0, 0, -vElliptischS2, 0, 250000.0, 9, Color.CYAN);

		// Stern S62 (kommt extrem nah an das Schwarze Loch heran)
		double distS62 = 200.0;
		double vKreisS62 = Math.sqrt(G * massSgrA / distS62);
		double vElliptischS62 = vKreisS62 * 0.30; // Nur 30% Speed -> stürzt fast hinein
		MasseKugel sternS62 = new MasseKugel(0, distS62, 0, -vElliptischS62, 0, 0, 180000.0, 8, Color.MAGENTA);


		// --- Das "Retter-System" (Hypothetische Habitable Randzone) ---
		// Ein stabiles Sternensystem weit genug entfernt vom chaotischen Zentrum, 
		// um intern Leben zu ermöglichen, während es das Zentrum umkreist.
		double distArcheSonne = 1200.0;
		double vArcheSonne = Math.sqrt(G * massSgrA / distArcheSonne);
		MasseKugel archeSonne = new MasseKugel(distArcheSonne, 0, 0, 0, -vArcheSonne, 0, 333030.0, 7, Color.YELLOW);

		double distArchePlanet = 15.0;
		// Die Orbitalgeschwindigkeit des Planeten um seine Sonne
		double vArchePlanet = Math.sqrt(G * archeSonne.mass / distArchePlanet);
		// Der Planet erbt die Bewegungsrichtung der Sonne und addiert seine eigene Orbitalgeschwindigkeit
		MasseKugel archePlanet = new MasseKugel(distArcheSonne + distArchePlanet, 0, 0, 0, -vArcheSonne - vArchePlanet, 0, 1.0, 4, Color.GREEN);

		bodies.add(sgrA);
		bodies.add(sternS2);
		bodies.add(sternS62);
		bodies.add(archeSonne);
		bodies.add(archePlanet);
	}
	
	public void elementarBausteinBasis() {
		// --- Der logisch optimale Elementar-Baustein (Das Gravitations-Oktett) ---
		// Repräsentiert die perfekte Symmetrie, die aus unendlicher Entropie-Angleichung entsteht.

		// 1. Das Zentrum (Analogie zum Atomkern oder supermassiven Attraktor)
		double massZentrum = 1000000;
		MasseKugel zentrum = new MasseKugel(0, 0, 0, 0, 0, 0, massZentrum, 18, Color.WHITE);
		bodies.add(zentrum);

		// 2. Die innere Schale (Analogie zum s-Orbital: 2 Massenpunkte)
		// Perfekt ausbalanciert, 180 Grad zueinander versetzt.
		double distInner = 25.0;
		double massInner = 10.0;
		// Bahngeschwindigkeit: v = sqrt(G * M / r)
		double vInner = Math.sqrt(G * massZentrum / distInner);
		
		for (int i = 0; i < 2; i++) {
			double angle = i * Math.PI; // 0 und 180 Grad
			double x = Math.cos(angle) * distInner;
			double y = Math.sin(angle) * distInner;
			double vx = -Math.sin(angle) * vInner;
			double vy = Math.cos(angle) * vInner;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, massInner, 5, new Color(255, 100, 100)));
		}

		// 3. Die äußere Schale (Analogie zu den 6 Entropieverknüpfungen / p-Orbitalen)
		// 6 Massenpunkte bilden ein perfektes Hexagon, das das Zentrum umkreist.
		double distOuter = 70.0;
		double massOuter = 15.0;
		
		// Korrektur der Orbitalgeschwindigkeit durch die Masse der inneren Schale
		// (Die äußeren Planeten spüren die Gravitation des Zentrums PLUS die der inneren Planeten)
		double effektiveZentrumsMasse = massZentrum + (2 * massInner);
		double vOuter = Math.sqrt(G * effektiveZentrumsMasse / distOuter);
		
		for (int i = 0; i < 6; i++) {
			double angle = i * (Math.PI / 3.0); // 60 Grad Abstand (360° / 6)
			double x = Math.cos(angle) * distOuter;
			double y = Math.sin(angle) * distOuter;
			double vx = -Math.sin(angle) * vOuter;
			double vy = Math.cos(angle) * vOuter;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, massOuter, 7, new Color(100, 200, 255)));
		}
	}
	
	
	public void protonenBasis() {
		// --- Das Matrix-Proton: Subatomare Skalierung durch Newtonsche Axiome ---
		
		// Skalierte Basis-Massen für die Engine
		double masseElektron = 12.0; 
		double masseElementar = masseElektron / 6.0; // Dein 1/6 Basisbaustein (2.0)
		
		// Das Proton besteht grob aus 3 Quarks (Up, Up, Down).
		// Wir teilen einen Großteil der Protonenmasse auf diese 3 massiven Zentren auf.
		double masseQuark = 1836.0 * masseElektron / 3.0; 
		
		double distQuarks = 8.0; // Extrem nah beieinander
		
		// Orbitalgeschwindigkeit für ein Dreikörper-System gleicher Masse auf einer Kreisbahn:
		// v = sqrt(G * M / (sqrt(3) * R))
		double vQuark = Math.sqrt(G * masseQuark / (Math.sqrt(3) * distQuarks));

		// 1. Der "Kern" (3 Quarks in perfekter Symmetrie, 120 Grad versetzt)
		for (int i = 0; i < 3; i++) {
			double angle = i * (2.0 * Math.PI / 3.0);
			double x = Math.cos(angle) * distQuarks;
			double y = Math.sin(angle) * distQuarks;
			double vx = -Math.sin(angle) * vQuark;
			double vy = Math.cos(angle) * vQuark;
			
			// Quarks werden in Rot/Blau/Grün dargestellt (Analogie zur Farbladung in der Quantenchromodynamik)
			Color qColor = (i == 0) ? Color.RED : (i == 1) ? Color.GREEN : Color.BLUE;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, masseQuark, 10, qColor));
		}

		// 2. Die Elementar-Schale (Die 1/6 Elektronen-Bausteine)
		// Wir lassen eine Wolke dieser Teilchen das Dreikörper-Zentrum in verschiedenen Orbits umkreisen.
		int anzahlElementar = 24; // Ein repräsentativer Schwarm (Skalierung für Performance)
		double zentrumsMasseGesamt = 3 * masseQuark;
		
		for (int i = 0; i < anzahlElementar; i++) {
			// Zufällige, aber stabile Abstände (Orbitalschalen)
			double distOrbital = 40.0 + (Math.random() * 80.0);
			double angle = Math.random() * 2.0 * Math.PI;
			
			double x = Math.cos(angle) * distOrbital;
			double y = Math.sin(angle) * distOrbital;
			
			// Orbitalgeschwindigkeit basiert auf der Gesamtmasse des Kerns
			double vOrbital = Math.sqrt(G * zentrumsMasseGesamt / distOrbital);
			
			// Leicht elliptische Fluktuationen erzeugen (+- 5% Speed) für Entropie
			double vFluktuation = vOrbital * (0.95 + Math.random() * 0.1);
			
			double vx = -Math.sin(angle) * vFluktuation;
			double vy = Math.cos(angle) * vFluktuation;
			
			// Wir übergeben deinen 1/6 Baustein
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, masseElementar, 3, Color.WHITE));
		}
	}
	
	
	
	public void orbitalRecyclerBasis() {
		// --- Kessler-Syndrom & Orbital-Recycling ---
		// Simulation eines Planeten mit gefährlichem Weltraumschrott und einer Recycler-Drohne.

		double erdMasse = 500000;
		// Der Heimatplanet
		MasseKugel erde = new MasseKugel(0, 0, 0, 0, 0, 0, erdMasse, 20, Color.BLUE);
		bodies.add(erde);

		// Weltraumschrott (hohe Entropie, extrem kreuzende, chaotische Bahnen)
		int schrottAnzahl = 50;
		for (int i = 0; i < schrottAnzahl; i++) {
			double dist = 60.0 + (Math.random() * 150.0);
			double angle = Math.random() * 2 * Math.PI;
			double vKreis = Math.sqrt(G * erdMasse / dist);
			
			// Elliptische und gegenläufige Geschwindigkeiten (0.4 bis 1.4 facher Speed)
			double vx = -Math.sin(angle) * vKreis * (0.4 + Math.random());
			double vy = Math.cos(angle) * vKreis * (0.4 + Math.random());
			
			MasseKugel schrott = new MasseKugel(Math.cos(angle)*dist, Math.sin(angle)*dist, 0, vx, vy, 0, 0.5, 2, Color.LIGHT_GRAY);
			bodies.add(schrott);
		}

		// Autonome / Steuerbare Recycler-Drohne
		// Startet in einem sicheren, weiten und perfekten Orbit
		double distRecycler = 280.0;
		double vRecycler = Math.sqrt(G * erdMasse / distRecycler);
		MasseKugel recycler = new MasseKugel(distRecycler, 0, 0, 0, -vRecycler, 0, 8.0, 6, Color.GREEN);
		
		bodies.add(recycler);
	}

	private void rotateBodyAroundDegrees(MasseKugel body, MasseKugel center, double degrees) {
		double theta = Math.toRadians(degrees);
		rotateBodyAround(body, center, theta);
	}

	private void rotateBodyAround(MasseKugel body, MasseKugel center, double theta) {
		double dx = body.x - center.x;
		double dy = body.y - center.y;
		double cos = Math.cos(theta);
		double sin = Math.sin(theta);
		double newDx = dx * cos - dy * sin;
		double newDy = dx * sin + dy * cos;
		body.x = center.x + newDx;
		body.y = center.y + newDy;
		double vx = body.vx;
		double vy = body.vy;
		body.vx = vx * cos - vy * sin;
		body.vy = vx * sin + vy * cos;
	}

	// NEU: Korrigierte Methode zur Berechnung der potentiellen Gesamtenergie
	// Sie summiert die potenzielle Energie zwischen *allen einzigartigen Paaren*
	// von Himmelskörpern.
	public double getEPotGes() {
		double totalEPot = 0;
		for (int i = 0; i < bodies.size(); i++) {
			MasseKugel b1 = bodies.get(i);
			for (int j = i + 1; j < bodies.size(); j++) { // Schleife über einzigartige Paare (j beginnt bei i+1)
				MasseKugel b2 = bodies.get(j);

				double dx = b2.x - b1.x;
				double dy = b2.y - b1.y;
				double dz = b2.z - b1.z; // Berücksichtigt die Z-Koordinate für den Abstand in 3D

				double r = Math.sqrt(dx * dx + dy * dy + dz * dz);

				// Behandle extrem kleine Abstände, um Division durch Null oder sehr große Werte
				// zu vermeiden
				// Schwellenwert (0.001) kann bei Bedarf an die Simulationsskala angepasst
				// werden
				if (r < 0.001) {
					r = 0.001;
				}

				totalEPot += -G * b1.mass * b2.mass / r;
			}
		}
		return totalEPot;
	}

		mp1.x -= flow1x;
		mp1.x += flow2x;
		mp2.x -= flow2x;
		mp2.x += flow1x;

		double flow1y = mp1.y * Math.random();
		double flow2y = mp2.y * Math.random();
		mp1.y -= flow1x;
		mp1.y += flow2x;
		mp2.y -= flow2x;
		mp2.y += flow1x;

		double flow1z = mp1.z * Math.random();

		mp1.z -= flow1x;
		mp1.z += flow2x;
		mp2.z -= flow2x;
		mp2.z += flow1x;
	}

	public void Gravity2D() {
		int n = bodies.size();
		double[] ax = new double[n];
		double[] ay = new double[n];
		

		for (int i = 0; i < n; i++) {
			MasseKugel bi = bodies.get(i);
			ax[i] = ay[i] = 0;
			if (!bi.movable)
				continue;
			for (int j = 0; j < n; j++) {
				if (i == j)
					continue;
				MasseKugel bj = bodies.get(j);
				double dx = bj.x - bi.x;
				double dy = bj.y - bi.y;
				double dz = bj.z - bi.z;
				
				if(dz!=0)continue;

				double r2 = dx * dx + dy * dy;
				double r = Math.sqrt(r2);

				double normdx = dx / r;
				double normdy = dy / r;
				

				ax[i] += G * bj.mass * normdx / r;
				ay[i] += G * bj.mass * normdy / r;
				
			}
		}
		// Positions- und Geschwindigkeitsupdate
		for (int i = 0; i < n; i++) {
			MasseKugel b = bodies.get(i);
			if (b.movable) {
				b.vx += ax[i];
				b.vy += ay[i];
				
			}
			b.updatePosition(1.0);
		}
	}

				double normdx = dx / r;

				double normdz = dz / r;

				ax[i] += G * bj.mass * normdx / Math.pow(r, verknüpfungen * 0.5 - 1);
				ay[i] += G * bj.mass * normdy / Math.pow(r, verknüpfungen * 0.5 - 1);
				az[i] += G * bj.mass * normdz / Math.pow(r, verknüpfungen * 0.5 - 1);
				
				if(ax[i]>r/2)ax[i]=r/2;
				if(ay[i]>r/2)ay[i]=r/2;
				if(az[i]>r/2)az[i]=r/2;
				
				if(ax[i]<-r/2)ax[i]=-r/2;
				if(ay[i]<-r/2)ay[i]=-r/2;
				if(az[i]<-r/2)az[i]=-r/2;
			}
		}
		// Positions- und Geschwindigkeitsupdate
		for (int i = 0; i < n; i++) {
			MasseKugel b = bodies.get(i);
			if (b.movable) {
				b.vx += ax[i];
				b.vy += ay[i];
				b.vz += az[i];
			}
			b.updatePosition(1.0);
		}
	}

	public void kinetischeFusion() {
		// --- Simulation von kinetischer Kernverschmelzung (Exotherm) ---
		List<MasseKugel> toRemove = new ArrayList<>();
		List<MasseKugel> toAdd = new ArrayList<>();

		for (int i = 0; i < bodies.size(); i++) {
			MasseKugel b1 = bodies.get(i);
			if (toRemove.contains(b1)) continue;

			for (int j = i + 1; j < bodies.size(); j++) {
				MasseKugel b2 = bodies.get(j);
				if (toRemove.contains(b2)) continue;

				double dx = b2.x - b1.x;
				double dy = b2.y - b1.y;
				double dz = b2.z - b1.z;
				double dist = Math.sqrt(dx*dx + dy*dy + dz*dz);

				// Kollisionsprüfung
				if (dist < (b1.radius + b2.radius)) {
					// Relativgeschwindigkeit ermitteln
					double dvx = b2.vx - b1.vx;
					double dvy = b2.vy - b1.vy;
					double dvz = b2.vz - b1.vz;
					
					// Skalare Geschwindigkeit im Quadrat (als Repräsentant der kinetischen Aufprallenergie)
					double vRelSq = dvx*dvx + dvy*dvy + dvz*dvz;

					// Schwellenwert: Nur wenn sie schnell genug aufeinanderprallen, fusionieren sie
					if (vRelSq > 2.0) { // Wert anpassbar für das Game-Balancing
						
						double neueMasse = b1.mass + b2.mass;
						
						// Völlig inelastischer Stoß: Impulserhaltung (p = m*v)
						double neuVx = (b1.mass * b1.vx + b2.mass * b2.vx) / neueMasse;
						double neuVy = (b1.mass * b1.vy + b2.mass * b2.vy) / neueMasse;
						double neuVz = (b1.mass * b1.vz + b2.mass * b2.vz) / neueMasse;
						
						// Das neu fusionierte, schwerere Element (z.B. Eisen)
						MasseKugel fusioniert = new MasseKugel(
								(b1.x + b2.x) / 2, (b1.y + b2.y) / 2, (b1.z + b2.z) / 2, 
								neuVx, neuVy, neuVz, 
								neueMasse, Math.max(b1.radius, b2.radius) + 1, Color.ORANGE);
						
						toRemove.add(b1);
						toRemove.add(b2);
						toAdd.add(fusioniert);
					}
				}
			}
		}
		// Listen aktualisieren
		bodies.removeAll(toRemove);
		bodies.addAll(toAdd);
	}
	
	
	public void Gravity(int verknüpfungen) {
		int n = bodies.size();

		for (int i = 0; i < n; i++) {
			MasseKugel bi = bodies.get((int)(Math.random()*n));

			
				MasseKugel bj = bodies.get((int)(Math.random()*n));
				if (bi == bj)
					continue;
				double dx = bj.x - bi.x;
				double dy = bj.y - bi.y;
				double dz = bj.z - bi.z;
				double r2 = dx * dx + dy * dy + dz * dz;
				double r = Math.sqrt(r2);

				bi.vx += G * bj.mass * dx / Math.pow(r, verknüpfungen / 0.5 - 1);
				bi.vy += G * bj.mass * dy / Math.pow(r, verknüpfungen / 0.5 - 1);
				bi.vz += G * bj.mass * dz / Math.pow(r, verknüpfungen / 0.5 - 1);

				bj.vx -= G * bi.mass * dx / Math.pow(r, verknüpfungen / 0.5-1);
				bj.vy -= G * bi.mass * dy / Math.pow(r, verknüpfungen / 0.5-1);
				bj.vz -= G * bi.mass * dz / Math.pow(r, verknüpfungen / 0.5-1);
			
			bi.updatePosition(1.0);
			bj.updatePosition(1.0);
		}
	}

	public void EntrophieSpeed(double alpha) {
		int n = bodies.size();
		if (n < 2)
			return;

		for (int i = 0; i < n; i++) {
			MasseKugel body1 = bodies.get(i);

			// Zufällige Auswahl eines zweiten Körpers für die Interaktion
			int j = (int) (Math.random() * n);
			while (i == j) { // Sicherstellen, dass nicht derselbe Körper gewählt wird
				j = (int) (Math.random() * n);
			}
			MasseKugel body2 = bodies.get(j);

			// Berechnung der mittleren Geschwindigkeit für die Angleichung
			double avg_vx = (body1.vx + body2.vx) / 2;
			double avg_vy = (body1.vy + body2.vy) / 2;
			double avg_vz = (body1.vz + body2.vz) / 2;

			// Entropie-basierte Geschwindigkeitsanpassung
			// Die Geschwindigkeiten werden proportional zur Differenz zur mittleren
			// Geschwindigkeit angepasst.
			// Die Konstante alpha steuert die Stärke der Angleichung.
			body1.vx += (avg_vx - body1.vx) * alpha;
			body1.vy += (avg_vy - body1.vy) * alpha;
			body1.vz += (avg_vz - body1.vz) * alpha;

			body2.vx += (avg_vx - body2.vx) * alpha;
			body2.vy += (avg_vy - body2.vy) * alpha;
			body2.vz += (avg_vz - body2.vz) * alpha;
		}
	}

	public void doppelsternBasis() {
		// --- Doppelsternsystem (Zirkumbinäres System) ---
		// Eine Simulation von zwei Sternen, die einander umkreisen (Baryzentrum),
		// mit inneren Planeten und einem äußeren "Matrix-Wächter"-Planeten.

		double massSternA = 380000; // Etwas massereicher als unsere Sonne
		double massSternB = 220000; // Etwas masseärmer (z.B. Roter Zwerg)
		double gesamtMasseSterne = massSternA + massSternB;

		double abstandSterne = 80.0; // Distanz der Sterne zueinander

		// 1. Berechnung des gemeinsamen Schwerpunktes (Baryzentrum)
		double rA = abstandSterne * (massSternB / gesamtMasseSterne);
		double rB = abstandSterne * (massSternA / gesamtMasseSterne);

		// 2. Berechnung der Orbitalgeschwindigkeiten der Sterne
		// Basierend auf Newtons Gravitationsgesetz
		double vOrbitalSterne = Math.sqrt(G * gesamtMasseSterne / abstandSterne);
		double vA = vOrbitalSterne * (rA / abstandSterne);
		double vB = vOrbitalSterne * (rB / abstandSterne);

		// --- Himmelskörper definieren ---

		// Stern A (Zentrum links)
		MasseKugel sternA = new MasseKugel(-rA, 0, 0, 0, -vA, 0, massSternA, 12, new Color(255, 220, 50));

	public void galaktischesZentrumBasis() {
		// --- Das Galaktische Zentrum ---
		// Eine Simulation des supermassereichen Schwarzen Lochs (Sagittarius A*)
		// und extremen stellaren Orbits.

		// Skalierte Masse für Sagittarius A* (Real ca. 4 Millionen Sonnenmassen)
		// Die Masse wird hier so skaliert, dass sie in die bestehende G-Konstante
		// und die Timesteps der Simulation passt, ohne Floating-Point-Overflows zu
		// erzeugen.
		double massSgrA = 1.5e9;

		// Das Schwarze Loch ruht (näherungsweise) im Zentrum
		MasseKugel sgrA = new MasseKugel(0, 0, 0, 0, 0, 0, massSgrA, 15, Color.BLACK);

		// --- S-Sterne (Extrem elliptische Orbits) ---
		// Um elliptische Bahnen zu erzeugen, geben wir den Sternen an ihrem fernsten
		// Punkt (Apoapsis)
		// eine Geschwindigkeit, die deutlich UNTER der perfekten
		// Kreisbahngeschwindigkeit liegt.

		// Stern S2 (ein realer Stern, der Sgr A* umkreist)
		double distS2 = 350.0;
		double vKreisS2 = Math.sqrt(G * massSgrA / distS2);
		double vElliptischS2 = vKreisS2 * 0.45; // Nur 45% Speed -> starke Exzentrizität
		MasseKugel sternS2 = new MasseKugel(-distS2, 0, 0, 0, -vElliptischS2, 0, 250000.0, 9, Color.CYAN);

		// Stern S62 (kommt extrem nah an das Schwarze Loch heran)
		double distS62 = 200.0;
		double vKreisS62 = Math.sqrt(G * massSgrA / distS62);
		double vElliptischS62 = vKreisS62 * 0.30; // Nur 30% Speed -> stürzt fast hinein
		MasseKugel sternS62 = new MasseKugel(0, distS62, 0, -vElliptischS62, 0, 0, 180000.0, 8, Color.MAGENTA);

		// --- Das "Retter-System" (Hypothetische Habitable Randzone) ---
		// Ein stabiles Sternensystem weit genug entfernt vom chaotischen Zentrum,
		// um intern Leben zu ermöglichen, während es das Zentrum umkreist.
		double distArcheSonne = 1200.0;
		double vArcheSonne = Math.sqrt(G * massSgrA / distArcheSonne);
		MasseKugel archeSonne = new MasseKugel(distArcheSonne, 0, 0, 0, -vArcheSonne, 0, 333030.0, 7, Color.YELLOW);

		double distArchePlanet = 15.0;

		double vArchePlanet = Math.sqrt(G * archeSonne.mass / distArchePlanet);

		// Orbitalgeschwindigkeit
		MasseKugel archePlanet = new MasseKugel(distArcheSonne + distArchePlanet, 0, 0, 0, -vArcheSonne - vArchePlanet,
				0, 1.0, 4, Color.GREEN);

		// Hinzufügen der Objekte zur Render-Engine
		bodies.add(sgrA);
		bodies.add(sternS2);
		bodies.add(sternS62);
		bodies.add(archeSonne);
		bodies.add(archePlanet);
	}

	public void elementarBausteinBasis() {
		// --- Der logisch optimale Elementar-Baustein (Das Gravitations-Oktett) ---
		// Repräsentiert die perfekte Symmetrie, die aus unendlicher
		// Entropie-Angleichung entsteht.

		// 1. Das Zentrum (Analogie zum Atomkern oder supermassiven Attraktor)
		double massZentrum = 1000000;
		MasseKugel zentrum = new MasseKugel(0, 0, 0, 0, 0, 0, massZentrum, 18, Color.WHITE);
		bodies.add(zentrum);

		// 2. Die innere Schale (Analogie zum s-Orbital: 2 Massenpunkte)
		// Perfekt ausbalanciert, 180 Grad zueinander versetzt.
		double distInner = 25.0;
		double massInner = 10.0;
		// Bahngeschwindigkeit: v = sqrt(G * M / r)
		double vInner = Math.sqrt(G * massZentrum / distInner);

		for (int i = 0; i < 2; i++) {
			double angle = i * Math.PI; // 0 und 180 Grad
			double x = Math.cos(angle) * distInner;
			double y = Math.sin(angle) * distInner;
			double vx = -Math.sin(angle) * vInner;
			double vy = Math.cos(angle) * vInner;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, massInner, 5, new Color(255, 100, 100)));
		}

		// 3. Die äußere Schale (Analogie zu den 6 Entropieverknüpfungen / p-Orbitalen)
		// 6 Massenpunkte bilden ein perfektes Hexagon, das das Zentrum umkreist.
		double distOuter = 70.0;
		double massOuter = 15.0;

		// Korrektur der Orbitalgeschwindigkeit durch die Masse der inneren Schale
		// (Die äußeren Planeten spüren die Gravitation des Zentrums PLUS die der
		// inneren Planeten)
		double effektiveZentrumsMasse = massZentrum + (2 * massInner);
		double vOuter = Math.sqrt(G * effektiveZentrumsMasse / distOuter);

		for (int i = 0; i < 6; i++) {
			double angle = i * (Math.PI / 3.0); // 60 Grad Abstand (360° / 6)
			double x = Math.cos(angle) * distOuter;
			double y = Math.sin(angle) * distOuter;
			double vx = -Math.sin(angle) * vOuter;
			double vy = Math.cos(angle) * vOuter;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, massOuter, 7, new Color(100, 200, 255)));
		}
	}

	public void protonenBasis() {
		// --- Das Matrix-Proton: Subatomare Skalierung durch Newtonsche Axiome ---

		// Skalierte Basis-Massen für die Engine
		double masseElektron = 12.0;
		double masseElementar = masseElektron / 6.0; // Dein 1/6 Basisbaustein (2.0)

		// Das Proton besteht grob aus 3 Quarks (Up, Up, Down).
		// Wir teilen einen Großteil der Protonenmasse auf diese 3 massiven Zentren auf.
		double masseQuark = 1836.0 * masseElektron / 3.0;

		double distQuarks = 8.0; // Extrem nah beieinander

		// Orbitalgeschwindigkeit für ein Dreikörper-System gleicher Masse auf einer
		// Kreisbahn:
		// v = sqrt(G * M / (sqrt(3) * R))
		double vQuark = Math.sqrt(G * masseQuark / (Math.sqrt(3) * distQuarks));

		// 1. Der "Kern" (3 Quarks in perfekter Symmetrie, 120 Grad versetzt)
		for (int i = 0; i < 3; i++) {
			double angle = i * (2.0 * Math.PI / 3.0);
			double x = Math.cos(angle) * distQuarks;
			double y = Math.sin(angle) * distQuarks;
			double vx = -Math.sin(angle) * vQuark;
			double vy = Math.cos(angle) * vQuark;

			// Quarks werden in Rot/Blau/Grün dargestellt (Analogie zur Farbladung in der
			// Quantenchromodynamik)
			Color qColor = (i == 0) ? Color.RED : (i == 1) ? Color.GREEN : Color.BLUE;
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, masseQuark, 10, qColor));
		}

		// 2. Die Elementar-Schale (Die 1/6 Elektronen-Bausteine)
		// Wir lassen eine Wolke dieser Teilchen das Dreikörper-Zentrum in verschiedenen
		// Orbits umkreisen.
		int anzahlElementar = 24; // Ein repräsentativer Schwarm (Skalierung für Performance)
		double zentrumsMasseGesamt = 3 * masseQuark;

		for (int i = 0; i < anzahlElementar; i++) {
			// Zufällige, aber stabile Abstände (Orbitalschalen)
			double distOrbital = 40.0 + (Math.random() * 80.0);
			double angle = Math.random() * 2.0 * Math.PI;

			double x = Math.cos(angle) * distOrbital;
			double y = Math.sin(angle) * distOrbital;

			// Orbitalgeschwindigkeit basiert auf der Gesamtmasse des Kerns
			double vOrbital = Math.sqrt(G * zentrumsMasseGesamt / distOrbital);

			// Leicht elliptische Fluktuationen erzeugen (+- 5% Speed) für Entropie
			double vFluktuation = vOrbital * (0.95 + Math.random() * 0.1);

			double vx = -Math.sin(angle) * vFluktuation;
			double vy = Math.cos(angle) * vFluktuation;

			// Wir übergeben deinen 1/6 Baustein
			bodies.add(new MasseKugel(x, y, 0, vx, vy, 0, masseElementar, 3, Color.WHITE));
		}
	}

		// Recycler-Drohne.

		double erdMasse = 500000;
		// Der Heimatplanet
		MasseKugel erde = new MasseKugel(0, 0, 0, 0, 0, 0, erdMasse, 20, Color.BLUE);
		bodies.add(erde);

		// Weltraumschrott (hohe Entropie, extrem kreuzende, chaotische Bahnen)
		int schrottAnzahl = 50;
		for (int i = 0; i < schrottAnzahl; i++) {
			double dist = 60.0 + (Math.random() * 150.0);
			double angle = Math.random() * 2 * Math.PI;
			double vKreis = Math.sqrt(G * erdMasse / dist);

			// Elliptische und gegenläufige Geschwindigkeiten (0.4 bis 1.4 facher Speed)
			double vx = -Math.sin(angle) * vKreis * (0.4 + Math.random());
			double vy = Math.cos(angle) * vKreis * (0.4 + Math.random());

			MasseKugel schrott = new MasseKugel(Math.cos(angle) * dist, Math.sin(angle) * dist, 0, vx, vy, 0, 0.5, 2,
					Color.LIGHT_GRAY);
			bodies.add(schrott);
		}

		// Autonome / Steuerbare Recycler-Drohne
		// Startet in einem sicheren, weiten und perfekten Orbit
		double distRecycler = 280.0;
		double vRecycler = Math.sqrt(G * erdMasse / distRecycler);
		MasseKugel recycler = new MasseKugel(distRecycler, 0, 0, 0, -vRecycler, 0, 8.0, 6, Color.GREEN);

		bodies.add(recycler);
	}

	private void rotateBodyAroundDegrees(MasseKugel body, MasseKugel center, double degrees) {
		double theta = Math.toRadians(degrees);
		rotateBodyAround(body, center, theta);
	}

	private void rotateBodyAround(MasseKugel body, MasseKugel center, double theta) {
		double dx = body.x - center.x;
		double dy = body.y - center.y;
		double cos = Math.cos(theta);
		double sin = Math.sin(theta);
		double newDx = dx * cos - dy * sin;
		double newDy = dx * sin + dy * cos;
		body.x = center.x + newDx;
		body.y = center.y + newDy;
		double vx = body.vx;
		double vy = body.vy;
		body.vx = vx * cos - vy * sin;
		body.vy = vx * sin + vy * cos;
	}

	// NEU: Korrigierte Methode zur Berechnung der potentiellen Gesamtenergie
	// Sie summiert die potenzielle Energie zwischen *allen einzigartigen Paaren*
	// von Himmelskörpern.
	public double getEPotGes() {
		double totalEPot = 0;
		for (int i = 0; i < bodies.size(); i++) {
			MasseKugel b1 = bodies.get(i);
			for (int j = i + 1; j < bodies.size(); j++) { // Schleife über einzigartige Paare (j beginnt bei i+1)
				MasseKugel b2 = bodies.get(j);

				double dx = b2.x - b1.x;
				double dy = b2.y - b1.y;
				double dz = b2.z - b1.z; // Berücksichtigt die Z-Koordinate für den Abstand in 3D

				double r = Math.sqrt(dx * dx + dy * dy + dz * dz);

				// Behandle extrem kleine Abstände, um Division durch Null oder sehr große Werte
				// zu vermeiden
				// Schwellenwert (0.001) kann bei Bedarf an die Simulationsskala angepasst
				// werden
				if (r < 0.001) {
					r = 0.001;
				}

				totalEPot += -G * b1.mass * b2.mass / r;
			}
		}
		return totalEPot;
	}

		mp1.x += flow2x;
		mp2.x -= flow2x;
		mp2.x += flow1x;

		double flow1y = mp1.y * Math.random();
		double flow2y = mp2.y * Math.random();
		mp1.y -= flow1x;
		mp1.y += flow2x;
		mp2.y -= flow2x;
		mp2.y += flow1x;

		double flow1z = mp1.z * Math.random();

		mp1.z += flow2x;

	public void GravityGemini(int foltern) {
		int n = bodies.size();
		double[] ax = new double[n];
		double[] ay = new double[n];
		double[] az = new double[n];

		for (int i = 0; i < n; i++) {
			MasseKugel bi = bodies.get(i);
			ax[i] = ay[i] = az[i] = 0;
			if (!bi.movable)
				continue;
			for (int j = 0; j < n; j++) {
				if (i == j)
					continue;
				MasseKugel bj = bodies.get(j);
				double dx = bj.x - bi.x;
				double dy = bj.y - bi.y;
				double dz = bj.z - bi.z;

				double r2 = dx * dx + dy * dy + dz * dz;
				double r = Math.sqrt(r2);

				if (r == 0)
					continue;

				double normdx = dx / r;
				double normdy = dy / r;
				double normdz = dz / r;

				ax[i] += G * bj.mass * normdx / Math.pow(r, verknüpfungen * 0.5 - 1);
				ay[i] += G * bj.mass * normdy / Math.pow(r, verknüpfungen * 0.5 - 1);
				az[i] += G * bj.mass * normdz / Math.pow(r, verknüpfungen * 0.5 - 1);

				if (ax[i] > r / 2)
					ax[i] = r / 2;
				if (ay[i] > r / 2)
					ay[i] = r / 2;
				if (az[i] > r / 2)
					az[i] = r / 2;

				if (ax[i] < -r / 2)
					ax[i] = -r / 2;
				if (ay[i] < -r / 2)
					ay[i] = -r / 2;
				if (az[i] < -r / 2)
					az[i] = -r / 2;
			}
		}
		// Positions- und Geschwindigkeitsupdate
		for (int i = 0; i < n; i++) {
			MasseKugel b = bodies.get(i);
			if (b.movable) {
				b.vx += ax[i];
				b.vy += ay[i];
				b.vz += az[i];
			}
			b.updatePosition(1.0);
		}
	}

	public void kinetischeFusion() {
		// --- Simulation von kinetischer Kernverschmelzung (Exotherm) ---
		List<MasseKugel> toRemove = new ArrayList<>();
		List<MasseKugel> toAdd = new ArrayList<>();

		for (int i = 0; i < bodies.size(); i++) {
			MasseKugel b1 = bodies.get(i);
			if (toRemove.contains(b1))
				continue;

			for (int j = i + 1; j < bodies.size(); j++) {
				MasseKugel b2 = bodies.get(j);
				if (toRemove.contains(b2))
					continue;

				double dx = b2.x - b1.x;
				double dy = b2.y - b1.y;
				double dz = b2.z - b1.z;
				double dist = Math.sqrt(dx * dx + dy * dy + dz * dz);

				// Kollisionsprüfung
				if (dist < (b1.radius + b2.radius)) {
					// Relativgeschwindigkeit ermitteln
					double dvx = b2.vx - b1.vx;
					double dvy = b2.vy - b1.vy;
					double dvz = b2.vz - b1.vz;

					// Skalare Geschwindigkeit im Quadrat (als Repräsentant der kinetischen
					// Aufprallenergie)
					double vRelSq = dvx * dvx + dvy * dvy + dvz * dvz;

					// Schwellenwert: Nur wenn sie schnell genug aufeinanderprallen, fusionieren sie
					if (vRelSq > 2.0) { // Wert anpassbar für das Game-Balancing

						double neueMasse = b1.mass + b2.mass;

						// Völlig inelastischer Stoß: Impulserhaltung (p = m*v)
						double neuVx = (b1.mass * b1.vx + b2.mass * b2.vx) / neueMasse;
						double neuVy = (b1.mass * b1.vy + b2.mass * b2.vy) / neueMasse;
						double neuVz = (b1.mass * b1.vz + b2.mass * b2.vz) / neueMasse;

						// Das neu fusionierte, schwerere Element (z.B. Eisen)
						MasseKugel fusioniert = new MasseKugel((b1.x + b2.x) / 2, (b1.y + b2.y) / 2, (b1.z + b2.z) / 2,
								neuVx, neuVy, neuVz, neueMasse, Math.max(b1.radius, b2.radius) + 1, Color.ORANGE);

						toRemove.add(b1);
						toRemove.add(b2);
						toAdd.add(fusioniert);
					}
				}
			}
		}
		// Listen aktualisieren
		bodies.removeAll(toRemove);
		bodies.addAll(toAdd);
	}

	public void Gravity(int verknüpfungen) {
		int n = bodies.size();

		for (int i = 0; i < n; i++) {
			MasseKugel bi = bodies.get((int) (Math.random() * n));

			MasseKugel bj = bodies.get((int) (Math.random() * n));
			if (bi == bj)
				continue;
			double dx = bj.x - bi.x;
			double dy = bj.y - bi.y;
			double dz = bj.z - bi.z;
			double r2 = dx * dx + dy * dy + dz * dz;
			double r = Math.sqrt(r2);

			bi.vx += G * bj.mass * dx / Math.pow(r, verknüpfungen / 0.5 - 1);
			bi.vy += G * bj.mass * dy / Math.pow(r, verknüpfungen / 0.5 - 1);
			bi.vz += G * bj.mass * dz / Math.pow(r, verknüpfungen / 0.5 - 1);

			bj.vx -= G * bi.mass * dx / Math.pow(r, verknüpfungen / 0.5 - 1);
			bj.vy -= G * bi.mass * dy / Math.pow(r, verknüpfungen / 0.5 - 1);
			bj.vz -= G * bi.mass * dz / Math.pow(r, verknüpfungen / 0.5 - 1);

			bi.updatePosition(1.0);
			bj.updatePosition(1.0);
		}
	}

    // Kraft in Newton, die auf beide MasseKugeln wirkt und diese wechselseitig in Richtung des anderen beschleunigt, so das die Gravitationskraft weiter steigt und der Abstand sinkt
    public double getGravityKraft(MasseKugel other) {
    	double kraft =0;
    	double distQ = (other.x-this.x)*(other.x-this.x)+(other.y-this.y)*(other.y-this.y)+(other.z-this.z)*(other.z-this.z);
    	kraft= G * this.mass*other.mass/distQ;
    	return kraft;
    }

    public void fusion(MasseKugel kugel2) {

    	double blue = color.getBlue()*massAlt+kugel2.color.getBlue()*massOther;
    	
    	red=Math.round(red/massNew);
    	green=Math.round(green/massNew);
    	blue=Math.round(blue/massNew);
    	
    	this.color=new Color((int)red,(int)green,(int)blue);

    public void moveTo(MasseKugel oponent) {
		double distQ = (this.x - oponent.x) * (this.x - oponent.x)
				+ (this.y - oponent.y) * (this.y - oponent.y)
				+ (this.z - oponent.z) * (this.z - oponent.z);
		double dist = Math.pow(distQ, 0.5);
		
		double dx=x-oponent.x;
		double dy=y-oponent.y;
		double dz=z-oponent.z;
		
		double movDist=Math.min(speed, Math.abs(dist-0));
		
		vx=-dx/dist*movDist;
		vy=-dy/dist*movDist;
		vz=-dz/dist*movDist;

	}
    
    public double getDistance(MasseKugel oponent) {
		double distQ = (this.x - oponent.y) * (this.x - oponent.x)
				+ (this.y - oponent.y) * (this.y - oponent.y)
				+ (this.z - oponent.z) * (this.z - oponent.z);
		double dist = Math.pow(distQ, 0.5);
		return dist;
	}
